{{- /* Renders data/talks.yaml as a grid of cards grouped by year. YouTube thumbnails are fetched at build time and served from this site, so visitors don't make requests to YouTube until they click through. */ -}} {{- $talks := hugo.Data.talks }} {{- $years := slice }} {{- range $talks }}{{ $years = $years | append .year }}{{ end }} {{- range sort ($years | uniq) "value" "desc" }} {{- $year := . }}

{{ $year }}

{{- range where $talks "year" $year }} {{- $href := "" }} {{- $kind := "Slides" }} {{- $icon := "file-lines" }} {{- $thumb := false }} {{- with .youtube }} {{- $href = printf "https://www.youtube.com/watch?v=%s" . }} {{- $kind = "Video" }} {{- $icon = "youtube" }} {{- $url := printf "https://i.ytimg.com/vi/%s/hqdefault.jpg" . }} {{- with try (resources.GetRemote $url) }} {{- with .Err }} {{- warnf "talks: failed to fetch thumbnail %s: %s" $url . }} {{- else with .Value }} {{- $thumb = .Fill "640x360 Center webp q80" }} {{- end }} {{- end }} {{- end }} {{- with .start }}{{ $href = printf "%s&t=%ds" $href . }}{{ end }} {{- with .video }}{{ $href = . }}{{ $kind = "Video" }}{{ $icon = "youtube" }}{{ end }} {{- with .podcast }}{{ $href = . }}{{ $kind = "Podcast" }}{{ $icon = "music" }}{{ end }} {{- if and (not $href) .links }}{{ $href = (index .links 0).url }}{{ end }}
{{- with $thumb }} {{- else }} {{ partial "icon.html" $icon }}{{ $kind }} {{- end }}

{{ .blurb }}
